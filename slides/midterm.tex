\begin{frame}{Problem}
  \textbf{Given} a video, \textbf{produce} a box around each pedestrian in every
  frame, with the \alert{same identity number} following the same person.
  \vspace{1.2em}

  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{Detection is not enough}

      {\small A detector answers ``where are the people'' separately in each
      frame. It has no way of knowing that the person in frame 10 is the person
      it saw in frame 9.}
    \column{0.47\textwidth}
      \textbf{Tracking adds identity}

      {\small We have to link detections across frames, through occlusion,
      through people crossing, and through frames where the detector misses
      someone entirely.}
  \end{columns}
  \vspace{1.0em}

  \begin{alertblock}{Why identity, and not just location}
    {\small Driving, surveillance, sports and robotics all need to know
    \emph{which} person, and for how long. Our target: track one pedestrian
    across an MOT16 sequence without losing them (rubric, grade A).}
  \end{alertblock}
\end{frame}

\begin{frame}{Detection: Faster R-CNN}
  \begin{columns}[T,onlytextwidth]
    \column{0.54\textwidth}
      \textbf{Why this model}
      \begin{itemize}\small
        \item The Region Proposal Network generates proposals inside the
              network, so no Selective Search
        \item Proposals and detection share one backbone, so each image is
              convolved once
        \item Good accuracy on small objects, and distant pedestrians are small
      \end{itemize}
      \vspace{0.5em}
      \textbf{Few-shot fine-tuning}

      {\small We freeze the ResNet50-FPN backbone and train only the detection
      heads, as the assignment suggests.}
    \column{0.42\textwidth}
      \begin{block}{Parameter budget}
        \begin{tabular}{lr}
          \toprule
          Total              & 41.3\,M \\
          Frozen             & 26.8\,M \\
          \textbf{Trainable} & \textbf{14.5\,M} \\
          \bottomrule
        \end{tabular}
        \vspace{0.5em}

        {\footnotesize Training 35\% of the parameters keeps the job inside
        Colab's memory budget.}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Association Problem}
  Each frame gives $N$ detections and we are holding $M$ tracks. Which detection
  continues which track?
  \vspace{0.9em}

  \begin{columns}[T,onlytextwidth]
    \column{0.45\textwidth}
      \textbf{Overlap alone fails}

      {\small When two people cross, each box overlaps the other's previous
      position, so position cannot separate them.}
      \vspace{0.6em}

      \textbf{Appearance alone fails}

      {\small Two people in similar clothing produce similar embeddings, and
      appearance drifts as someone turns.}
    \column{0.50\textwidth}
      \begin{block}{Blended cost}
        \vspace{-0.4em}
        \[
          C_{ij} = (1-\lambda)\underbrace{(1 - \mathrm{IoU}_{ij})}_{\text{position}}
                 + \lambda \underbrace{(1 - \cos(e_i, e_j))}_{\text{appearance}}
        \]
        \vspace{-0.5em}

        {\footnotesize Minimised with the Hungarian algorithm, giving a globally
        optimal one-to-one assignment.}
      \end{block}
  \end{columns}
  \vspace{0.6em}
  {\footnotesize Pairs failing the IoU or appearance gate are rejected before
  matching. A track survives \texttt{max\_age} unmatched frames, so a brief
  occlusion does not create a new identity.}
\end{frame}

\begin{frame}{Two traps in the annotations}
  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \begin{alertblock}{Coordinates are 1-indexed}
        The files count pixels from 1 and NumPy counts from 0. If you miss
        this, every box is off by one pixel. The output still looks correct,
        but IoU drops and the metrics get worse for no visible reason.
      \end{alertblock}
    \column{0.47\textwidth}
      \begin{alertblock}{Not every person is a target}
        Classes 7, 8 and 12 (static person, distractor, reflection) are people
        but are not scored. Training on them teaches the detector to fire on
        mannequins and reflections.
      \end{alertblock}
  \end{columns}
  \vspace{1.1em}
  Both are handled in \texttt{mot/data/mot16.py} and covered by unit tests. We
  wrote the tests first because neither failure produces an obvious symptom.
\end{frame}

\begin{frame}{Data Augmentation}
  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
      {\small MOT16 gives us few scenes, so the model can memorise those scenes
      instead of learning what a pedestrian looks like.}
      \vspace{0.6em}

      \textbf{Detection}
      \begin{itemize}\small
        \item Horizontal flip, boxes mirrored to match
        \item Colour jitter: brightness, contrast, saturation
      \end{itemize}
      \vspace{0.4em}
      \textbf{Re-ID crops}
      \begin{itemize}\small
        \item Random crop and horizontal flip
        \item Random erasing, simulating occlusion
      \end{itemize}
    \column{0.45\textwidth}
      \begin{block}{What we leave out}
        No vertical flips and no large rotations. Pedestrians are upright in
        every frame of every sequence, so an upside-down person is a pose the
        model will never see at test time.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Re-identification: a similarity model, not a classifier}
  {\small A classifier over the people in a video needs retraining every time
  somebody new walks in, and MOT16 is full of people walking in. Instead we
  learn an \emph{embedding}: crops of one person land close together, crops of
  different people land apart.}
  \vspace{0.4em}

  \centering
  \begin{tikzpicture}[node distance=0.65cm]
    \node[io]                              (a)    {Crop A};
    \node[io, below=1.2cm of a]            (b)    {Crop B};
    \node[stage, right=1.0cm of a]         (cnna) {Shared CNN};
    \node[stage, right=1.0cm of b]         (cnnb) {Shared CNN};
    \node[io, right=0.9cm of cnna]         (ea)   {$e_A$};
    \node[io, right=0.9cm of cnnb]         (eb)   {$e_B$};
    \node[stage, right=1.2cm of $(ea)!0.5!(eb)$] (d) {Distance\\$\Vert e_A - e_B\Vert$};

    \draw[flow] (a) -- (cnna);
    \draw[flow] (b) -- (cnnb);
    \draw[flow] (cnna) -- (ea);
    \draw[flow] (cnnb) -- (eb);
    \draw[flow] (ea) -- (d);
    \draw[flow] (eb) -- (d);
    \draw[<->, dashed, thick, accent] (cnna) --
      node[right, font=\scriptsize] {same weights} (cnnb);
  \end{tikzpicture}
  \vspace{1.0em}

  \begin{columns}[T,onlytextwidth]
    \column{0.41\textwidth}
      {\footnotesize Four convolution blocks, adaptive pooling, then a
      128-dimensional L2-normalised embedding. 1.5\,M parameters, so it trains
      in minutes.}
    \column{0.6\textwidth}
      \begin{block}{Contrastive loss}
        \vspace{-0.7em}
        \[
          L = \tfrac12\left[ y\,d^2 + (1-y)\max(0, m - d)^2 \right]
        \]
        \vspace{-0.7em}

        {\footnotesize Matching pairs are pulled together, non-matching pairs
        pushed past the margin $m$.}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{A problem we had to resolve}
  \begin{alertblock}{The course slides give four different crop sizes}
    {\small Training is specified at $16\times16$, inference at $24\times24$,
    retrieved crops at $30\times16$, and the reference \texttt{Siamese\_Network}
    code uses \texttt{nn.Linear(128*22*22, ...)}, needing roughly
    $102\times102$. No single size satisfies all four.}
  \end{alertblock}
  \vspace{0.7em}

  \begin{columns}[T,onlytextwidth]
    \column{0.47\textwidth}
      \textbf{What we chose}
      \begin{itemize}\small
        \item $128\times64$, the resolution Market-1501 normalises its own
              images to
        \item \texttt{AdaptiveAvgPool2d} instead of a hardcoded flatten
      \end{itemize}
    \column{0.47\textwidth}
      \textbf{Why}
      \begin{itemize}\small
        \item $16\times16$ cannot resolve clothing, which is the signal
              re-identification relies on
        \item Adaptive pooling accepts any input size, so the choice is not
              baked into the architecture
      \end{itemize}
  \end{columns}
\end{frame}

% ─────────────────────────────────────────── Sanjana: evaluation and demo
\begin{frame}{How we measure success}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{CLEAR-MOT metrics}
        \vspace{-0.5em}
        \[ \mathrm{MOTA} = 1 - \frac{\mathrm{FN} + \mathrm{FP} + \mathrm{IDSW}}{\text{ground truth boxes}} \]
        \vspace{-0.6em}
        \begin{itemize}\footnotesize
          \item \textbf{FN:} a person we failed to detect
          \item \textbf{FP:} a box with nobody in it
          \item \textbf{IDSW:} the ID changed on one person
        \end{itemize}
      \end{block}
      \vspace{0.2em}
      {\footnotesize \textbf{MOTP} is the mean IoU of matched boxes, so it
      measures how tightly the boxes fit.}
    \column{0.44\textwidth}
      \begin{block}{Why ID switches matter}
        A missed frame can be recovered on the next one. An ID switch means we
        have confused two people, which is the failure the re-identification
        component exists to prevent and the one the rubric grades.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Results: first end-to-end run}
  MOT16-02, 600 frames, on a Colab T4. COCO-pretrained detector, \textbf{no
  fine-tuning yet}; re-identification trained for 8 epochs on MOT16 crops.
  \vspace{0.2em}

  \begin{columns}[T,onlytextwidth]
    \column{0.5\textwidth}
      \begin{alertblock}{Against the rubric}
        {\small One real pedestrian is held by a single id for \textbf{218 of
        the 230 frames they appear in (94.8\%)}, before any fine-tuning.
        \vspace{0.1em}

        But the \textbf{median across 44 people is 22.4\%}. One person is
        tracked well; most are not.}
      \end{alertblock}
      \vspace{0.2em}
      {\footnotesize Measured against the ground-truth person, not whichever
      track lived longest. Re-ID accuracy 0.535 $\rightarrow$ 0.932, 4.0 FPS.}
    \column{0.48\textwidth}
      \begin{block}{Whole sequence}
        {\footnotesize
        \begin{tabular}{lr}
          \toprule
          MOTA            & 0.232 \\
          MOTP            & 0.792 \\
          \midrule
          True positives  & 7{,}004 \\
          False positives & 2{,}713 \\
          False negatives & 10{,}829 \\
          ID switches     & 151 \\
          \bottomrule
        \end{tabular}}
        \vspace{0.2em}

        {\footnotesize MOTP 0.792 means the boxes are tight. MOTA is held down
        almost entirely by false negatives.}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Does re-identification help? Not yet, and we know why}
  \begin{columns}[T,onlytextwidth]
    \column{0.44\textwidth}
      \begin{block}{Ablation, 600 frames}
        {\footnotesize
        \begin{tabular}{lrr}
          \toprule
                & with & without \\
          \midrule
          MOTA  & 0.232 & 0.232 \\
          MOTP  & 0.792 & 0.792 \\
          IDSW  & 151   & 148 \\
          \bottomrule
        \end{tabular}}
        \vspace{0.3em}

        {\footnotesize Appearance changes nothing.}
      \end{block}
    \column{0.52\textwidth}
      \begin{block}{Detection threshold sweep, 150 frames}
        {\footnotesize
        \begin{tabular}{rrrrr}
          \toprule
          thresh & MOTA & recall & FN & IDSW \\
          \midrule
          0.50 & $-$0.006 & 0.509 & 1{,}676 & 80 \\
          0.60 & 0.105 & 0.490 & 1{,}741 & 72 \\
          0.70 & 0.190 & 0.455 & 1{,}860 & 34 \\
          0.80 & \textbf{0.268} & 0.400 & 2{,}049 & 15 \\
          \bottomrule
        \end{tabular}}
      \end{block}
  \end{columns}
  \vspace{0.6em}

  \begin{alertblock}{What this tells us}
    {\footnotesize False negatives outnumber ID switches seventy to one, and
    lowering the threshold lifts recall only 0.40 to 0.51 while false positives
    quadruple. Recall is capped by people the COCO model never finds: small,
    distant, occluded. \textbf{Detection is the bottleneck, so association has
    nothing left to disambiguate.} That is the case for fine-tuning, and it is
    what we do next.}
  \end{alertblock}
\end{frame}
